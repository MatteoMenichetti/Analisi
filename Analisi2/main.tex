% Figura: rappresentazione nel piano dei punti di interesse
% per la dimostrazione del Teorema di Schwarz (caso n=2).
% Da includere con \input{Analisi2/figures/dimostrazione_schwarz_tikz.tex}
% dentro un ambiente figure, oppure compilare a parte per ottenere un PNG.
\begin{tikzpicture}[
    >=Latex,
    line width=0.85pt,
    font=\fontsize{10}{12}\selectfont
  ]

  % ---- parametri numerici dei punti di interesse ----
  \def\xzero{2.4}
  \def\xone{3.3}
  \def\xtwo{4.3}
  \def\xbig{5.8}

  \def\yzero{1.7}
  \def\yone{3.15}
  \def\ytwo{4.05}
  \def\ybig{5.4}

  % ---- insieme aperto A ----
  \draw[line width=1.15pt, line join=round]
    plot [smooth cycle, tension=0.85] coordinates {
      (-1.15,4.6) (0.15,6.15) (2.1,6.55) (4.15,6.2) (6.45,6.45)
      (7.35,4.9) (7.0,2.6) (7.45,0.3) (6.1,-1.35) (3.5,-1.55)
      (1.2,-1.05) (-0.9,-0.3) (-1.5,2.1)
    };

  % ---- assi cartesiani ----
  \draw (-1.15,0) -- (0,0);
  \draw[->] (0,0) -- (7.9,0);
  \draw (0,-1.15) -- (0,0);
  \draw[->] (0,0) -- (0,7.0);

  % ---- rettangolo "piccolo" fino a P0=(x0,y0) ----
  \draw (0,\yzero) -- (\xzero,\yzero) -- (\xzero,0);

  % ---- rettangolo "grande" fino a P=(x,y) ----
  \draw (0,\ybig) -- (\xbig,\ybig) -- (\xbig,0);

  % ---- tacche per x1, x2 sull'asse x ----
  \draw (\xone,-0.11) -- (\xone, 0.11);
  \draw (\xtwo,-0.11) -- (\xtwo, 0.11);

  % ---- tacche per y1, y2 sull'asse y ----
  \draw (-0.11,\yone) -- (0.11,\yone);
  \draw (-0.11,\ytwo) -- (0.11,\ytwo);

  % ---- punti P0 e P ----
  \fill (\xzero,\yzero) circle (1.6pt);
  \fill (\xbig,\ybig) circle (1.6pt);

  % ---- etichette sull'asse x ----
  \node[below=2pt] at (\xzero,0) {$x_0$};
  \node[below=2pt] at (\xone,0) {$x_1$};
  \node[below=2pt] at (\xtwo,0) {$x_2$};
  \node[below=2pt] at (\xbig,0) {$x$};

  % ---- etichette sull'asse y ----
  \node[left=2pt] at (0,\yzero) {$y_0$};
  \node[left=2pt] at (0,\yone) {$y_1$};
  \node[left=2pt] at (0,\ytwo) {$y_2$};
  \node[left=2pt] at (0,\ybig) {$y$};

  % ---- etichette dei punti ----
  \node[anchor=south west, xshift=2pt, yshift=1pt] at (\xzero,\yzero) {$P_0$};
  \node[anchor=south west, xshift=2pt, yshift=1pt] at (\xbig,\ybig) {$P$};

  % ---- etichetta dell'insieme aperto ----
  \node at (6.55,5.95) {$A$};

\end{tikzpicture}
